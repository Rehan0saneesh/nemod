import os
from fastapi import Header, HTTPException, status
from supabase import create_client, Client
from backend.config import settings

supabase: Client = create_client(settings.SUPABASE_URL, settings.SUPABASE_SERVICE_KEY)

async def get_current_user(authorization: str = Header(...)) -> str:
    if not authorization.startswith("Bearer "):
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Invalid authorization header format"
        )
    token = authorization.split(" ")[1]
    try:
        user_res = supabase.auth.get_user(token)
        if not user_res or not user_res.user:
            raise HTTPException(
                status_code=status.HTTP_401_UNAUTHORIZED,
                detail="User session invalid or expired"
            )
        return str(user_res.user.id)
    except Exception as e:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail=f"Token verification failed: {str(e)}"
        )

import json
import time
from openai import OpenAI
from backend.config import settings

class TriageAgent:
    def __init__(self, client: OpenAI):
        self.client = client

    def process(self, raw_manifest: str) -> tuple[dict, int]:
        start = time.perf_counter()
        prompt = (
            "You are a logistics network extraction parser. Convert the following unstructured "
            "delivery manifest, fleet coordinates, or stop list into a strict mathematical graph.\n"
            "Output JSON only with keys:\n"
            "  'nodes': list of integer IDs (0 to N-1)\n"
            "  'edges': list of [u, v, weight] lists (undirected, weight is distance/cost)\n\n"
            f"MANIFEST:\n{raw_manifest}"
        )
        response = self.client.chat.completions.create(
            model=settings.MODEL_NANO,
            messages=[
                {"role": "system", "content": "Return valid JSON only. Do not wrap in markdown tags."},
                {"role": "user", "content": prompt}
            ],
            response_format={"type": "json_object"},
            temperature=0.0
        )
        latency_ms = int((time.perf_counter() - start) * 1000)
        data = json.loads(response.choices[0].message.content)
        return data, latency_ms

        "nodes": graph_dict["nodes"],
        "edges": graph_dict["edges"],
        "p_layers": hamiltonian_spec.get("p_layers", 1),

    latency_ms = int((time.perf_counter() - start) * 1000)
    return res.json(), latency_ms

async def create_dispatch(req: CreateDispatchRequest, user_id: str = Depends(get_current_user)):
    insert_res = supabase.table("dispatch_runs").insert({
        "user_id": user_id,
        "title": req.title,
        "status": "RUNNING",
        "compute_target": req.compute_target,
        "raw_payload": {"manifest": req.manifest_text}
    }).execute()
    run_id = insert_res.data[0]["id"]

    try:
        graph_dict, nano_triage_ms = triage_agent.process(req.manifest_text)
        hamiltonian_spec, ultra_ms = hamiltonian_agent.process(graph_dict)

        supabase.table("dispatch_runs").update({
            "graph_data": graph_dict,
            "hamiltonian_spec": hamiltonian_spec,
            "telemetry": {
                "triage_ms": nano_triage_ms,
                "hamiltonian_ms": ultra_ms
            }
        }).eq("id", run_id).execute()

        if req.compute_target == "LOCAL_GPU":
            return {
                "run_id": run_id,
                "status": "DELEGATE_LOCAL",
                "graph_data": graph_dict,
                "hamiltonian_spec": hamiltonian_spec,
                "telemetry": {
                    "triage_ms": nano_triage_ms,
                    "hamiltonian_ms": ultra_ms
                }
            }
        else:
            solver_res, solver_ms = run_cloud_qaoa(graph_dict, hamiltonian_spec)
            bitstring = solver_res["best_bitstring"]
            solution_data, decode_ms = solution_agent.process(graph_dict, bitstring)

            total_telemetry = {
                "triage_ms": nano_triage_ms,
                "hamiltonian_ms": ultra_ms,
                "solver_ms": solver_ms,
                "decode_ms": decode_ms,
                "energy": solver_res.get("energy", 0.0)
            }

            supabase.table("dispatch_runs").update({
                "status": "COMPLETED",
                "optimal_bitstring": bitstring,
                "solution_json": solution_data,
                "telemetry": total_telemetry
            }).eq("id", run_id).execute()

            return {
                "run_id": run_id,
                "status": "COMPLETED",
                "bitstring": bitstring,
                "solution": solution_data,
                "telemetry": total_telemetry

        "solution_json": solution_data,
        "telemetry": existing_telemetry
    }).eq("id", run_id).execute()

import sys
import json
import time
import networkx as nx
import cudaq

def run_solver(graph_data: dict, p_layers: int, gamma_init: float, beta_init: float):
    start_time = time.perf_counter()

    try:
        cudaq.set_target("nvidia")
    except Exception:
        cudaq.set_target("default")

    G = nx.Graph()
    for u, v, w in graph_data["edges"]:
        G.add_edge(int(u), int(v), weight=float(w))

    num_qubits = len(graph_data["nodes"])
    p = p_layers

    hamiltonian = 0
    for u, v, data in G.edges(data=True):
        w = data.get("weight", 1.0)
        hamiltonian += 0.5 * w * (cudaq.spin.i - cudaq.spin.z(u) * cudaq.spin.z(v))

    @cudaq.kernel
    def qaoa_kernel(qubit_count: int, layers: int, thetas: list[float]):
        qreg = cudaq.qvector(qubit_count)
        h(qreg)
        for l in range(layers):
            gamma = thetas[l]
            beta = thetas[l + layers]
            for edge_idx, (u, v) in enumerate(G.edges()):
                cx(qreg[u], qreg[v])
                rz(2.0 * gamma, qreg[v])
                cx(qreg[u], qreg[v])
            for q in range(qubit_count):
                rx(2.0 * beta, qreg[q])

    optimizer = cudaq.optimizers.NelderMead()
    initial_thetas = [gamma_init] * p + [beta_init] * p

    def objective(params):
        return -cudaq.observe(qaoa_kernel, hamiltonian, num_qubits, p, params).expectation()

    _, opt_thetas = optimizer.optimize(dimensions=2 * p, function=objective)
    counts = cudaq.sample(qaoa_kernel, num_qubits, p, opt_thetas, shots_count=1000)

    latency_ms = int((time.perf_counter() - start_time) * 1000)

    return {
        "status": "success",
        "best_bitstring": counts.most_probable(),
        "energy": float(objective(opt_thetas)),
        "solver_latency_ms": latency_ms,
        "target": cudaq.get_target().name
    }

if __name__ == "__main__":
    raw_in = sys.stdin.read()
    payload = json.loads(raw_in)
    res = run_solver(
        payload["graph_data"],
        payload.get("p_layers", 1),
        payload.get("gamma_init", 0.5),
        payload.get("beta_init", 0.5)
    )
    print(json.dumps(res))

void AuthDialog::handleLogin() { performAuth("/auth/v1/token?grant_type=password"); }
void AuthDialog::handleRegister() { performAuth("/auth/v1/signup"); }

void AuthDialog::performAuth(const QString& endpoint) {
    QUrl url(supabaseUrl + endpoint);
    QNetworkRequest request(url);
    request.setHeader(QNetworkRequest::ContentTypeHeader, "application/json");
    request.setRawHeader("apikey", anonKey.toUtf8());

    QJsonObject body;
    body["email"] = emailEdit->text().trimmed();
    body["password"] = passwordEdit->text();

    QNetworkReply* reply = net->post(request, QJsonDocument(body).toJson());
    connect(reply, &QNetworkReply::finished, [this, reply]() {
        if (reply->error() == QNetworkReply::NoError) {
            auto doc = QJsonDocument::fromJson(reply->readAll());
            jwtToken = doc.object()["access_token"].toString();
            userEmail = emailEdit->text().trimmed();
            QMessageBox::information(this, "Success", "Authenticated successfully.");
            accept();
        } else {
            QMessageBox::warning(this, "Authentication Failed", reply->readAll());
        }
        reply->deleteLater();
    });
}

    req.setRawHeader("Authorization", ("Bearer " + token).toUtf8());

    QJsonObject body;
    body["title"] = title;
    body["manifest_text"] = manifest;
    body["compute_target"] = target;

    QNetworkReply* reply = net->post(req, QJsonDocument(body).toJson());
    connect(reply, &QNetworkReply::finished, [this, reply]() {
        if (reply->error() == QNetworkReply::NoError) {
            emit dispatchCreated(QJsonDocument::fromJson(reply->readAll()).object());
        } else {
            emit errorOccurred("Dispatch creation failed: " + reply->errorString());
        }
        reply->deleteLater();
    });
}

void CloudSyncClient::completeLocalRun(const QString& runId, const QString& bitstring, int latencyMs, double energy) {
    QNetworkRequest req(QUrl(gatewayUrl + "/api/v1/dispatch/" + runId + "/complete_local"));
    req.setHeader(QNetworkRequest::ContentTypeHeader, "application/json");
    req.setRawHeader("Authorization", ("Bearer " + token).toUtf8());

    QJsonObject body;
    body["bitstring"] = bitstring;
    body["solver_latency_ms"] = latencyMs;
    body["energy"] = energy;

    QNetworkReply* reply = net->post(req, QJsonDocument(body).toJson());
    connect(reply, &QNetworkReply::finished, [this, reply]() {
        if (reply->error() == QNetworkReply::NoError) {
            emit localRunCompleted(QJsonDocument::fromJson(reply->readAll()).object());
        } else {
            emit errorOccurred("Failed syncing local run: " + reply->errorString());
        }
        reply->deleteLater();
    });
}

LocalRunner::LocalRunner(QObject* parent) : QObject(parent), process(new QProcess(this)) {
    connect(process, &QProcess::finished, [this](int exitCode, QProcess::ExitStatus) {
        if (exitCode == 0) {
            auto out = process->readAllStandardOutput();
            emit finished(QJsonDocument::fromJson(out).object());
        } else {
            emit error(process->readAllStandardError());
        }
    });
}

void LocalRunner::runQAOA(const QJsonObject& graphData, int pLayers, double gamma, double beta) {
    QJsonObject payload;
    payload["graph_data"] = graphData;
    payload["p_layers"] = pLayers;
    payload["gamma_init"] = gamma;
    payload["beta_init"] = beta;

    QString scriptPath = "backend/solvers/local_cudaq.py";
    process->start("python3", QStringList() << scriptPath);
    process->write(QJsonDocument(payload).toJson());
    process->closeWriteChannel();
}

void MainWindow::onDispatchCreated(const QJsonObject& data) {
    currentRunId = data["run_id"].toString();

    metricsTable->setItem(0, 1, new QTableWidgetItem(QString::number(telem["triage_ms"].toInt()) + " ms"));
    metricsTable->setItem(1, 1, new QTableWidgetItem(QString::number(telem["hamiltonian_ms"].toInt()) + " ms"));

    if (data["status"].toString() == "DELEGATE_LOCAL") {
        appendLog("LOCAL_CUDA-Q", "Starting QAOA state-vector on local NVIDIA hardware...");
        auto hSpec = data["hamiltonian_spec"].toObject();
        localRunner->runQAOA(
            data["graph_data"].toObject(),
            hSpec["p_layers"].toInt(),
            hSpec["gamma_init"].toDouble(),
            hSpec["beta_init"].toDouble()
        );

        onSyncFinished(data);
    }
}

    metricsTable->setItem(4, 1, new QTableWidgetItem(res["best_bitstring"].toString()));

        currentRunId,
        res["best_bitstring"].toString(),
        res["solver_latency_ms"].toInt(),
        res["energy"].toDouble()
    );
}
